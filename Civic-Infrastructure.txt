    <script>


        // -----------------------------------
        // Sample complaints
        // -----------------------------------

        let complaints = [

            {
                id: 1,

                problem:
                    "Large pothole near the market",

                category:
                    "Roads",

                area:
                    "Model Town",

                description:
                    "A large pothole has appeared near the market and is making it difficult for two-wheelers to pass.",

                urgency:
                    "High",

                status:
                    "In Progress",

                supporters:
                    14

            },


            {
                id: 2,

                problem:
                    "Garbage not collected",

                category:
                    "Garbage",

                area:
                    "Green Park",

                description:
                    "Garbage has not been collected for several days and is starting to create a bad smell.",

                urgency:
                    "Medium",

                status:
                    "Reported",

                supporters:
                    6

            },


            {
                id: 3,

                problem:
                    "Streetlight not working",

                category:
                    "Streetlight",

                area:
                    "College Road",

                description:
                    "The streetlight has not been working for almost a week.",

                urgency:
                    "Medium",

                status:
                    "Resolved",

                supporters:
                    21

            }

        ];


        // -----------------------------------
        // Load saved data
        // -----------------------------------

        let savedData =
            localStorage.getItem("myComplaints");


        if (savedData) {

            complaints =
                JSON.parse(savedData);

        }


        // -----------------------------------
        // Save complaints
        // -----------------------------------

        function saveComplaints() {

            localStorage.setItem(
                "myComplaints",
                JSON.stringify(complaints)
            );

        }


        // -----------------------------------
        // Calculate community attention
        // -----------------------------------

        function getAttention(complaint) {

            let score =
                complaint.supporters;


            // High urgency gets more attention

            if (complaint.urgency === "High") {

                score = score + 10;

            }


            // Safety problems are important

            if (complaint.category === "Public Safety") {

                score = score + 5;

            }


            return score;

        }


        // -----------------------------------
        // Display complaints
        // -----------------------------------

        function displayComplaints() {

            let list =
                document.getElementById(
                    "complaintList"
                );


            let search =
                document.getElementById(
                    "search"
                ).value.toLowerCase();


            list.innerHTML = "";


            let found = false;


            complaints.forEach(
                function(complaint) {


                    let problem =
                        complaint.problem.toLowerCase();


                    let area =
                        complaint.area.toLowerCase();


                    // Search filter

                    if (
                        !problem.includes(search) &&
                        !area.includes(search)
                    ) {

                        return;

                    }


                    found = true;


                    // Decide status class

                    let statusClass =
                        "reported";


                    if (
                        complaint.status ===
                        "In Progress"
                    ) {

                        statusClass =
                            "progress";

                    }


                    if (
                        complaint.status ===
                        "Resolved"
                    ) {

                        statusClass =
                            "resolved";

                    }


                    // Create card

                    let card =
                        document.createElement(
                            "div"
                        );


                    card.className =
                        "complaint";


                    card.innerHTML = `

                        <div class="complaint-header">

                            <div class="complaint-title">

                                ${complaint.problem}

                            </div>

                            <div class="category">

                                ${complaint.category}

                            </div>

                        </div>


                        <p>

                            ${complaint.description}

                        </p>


                        <div class="information">

                            <span>
                                📍 ${complaint.area}
                            </span>

                            <span>
                                Priority:
                                ${complaint.urgency}
                            </span>

                            <span
                                class="status ${statusClass}"
                            >
                                ${complaint.status}
                            </span>

                        </div>


                        <div class="attention">

                            Community attention:

                            <strong>
                                ${getAttention(complaint)}
                            </strong>

                        </div>


                        <div class="actions">

                            <button
                                class="vote"
                                onclick="supportComplaint(${complaint.id})"
                            >
                                👍 Support
                                (${complaint.supporters})
                            </button>


                            <button
                                class="update"
                                onclick="showTimeline(${complaint.id})"
                            >
                                View Follow-up
                            </button>

                        </div>


                        <div
                            class="timeline"
                            id="timeline-${complaint.id}"
                        >

                            <p>
                                Report submitted
                            </p>

                            <p>
                                Community members can support this issue
                            </p>

                            <p>
                                Current status:
                                ${complaint.status}
                            </p>

                        </div>

                    `;


                    list.appendChild(card);

                }
            );


            if (!found) {

                list.innerHTML = `

                    <p>
                        No reports found.
                    </p>

                `;

            }


            updateStatistics();

        }


        // -----------------------------------
        // Submit new complaint
        // -----------------------------------

        document
            .getElementById("reportForm")
            .addEventListener(
                "submit",
                function(event) {


                    // Stop the page from refreshing

                    event.preventDefault();


                    // Get form values

                    let problem =
                        document.getElementById(
                            "problem"
                        ).value;


                    let category =
                        document.getElementById(
                            "category"
                        ).value;


                    let area =
                        document.getElementById(
                            "area"
                        ).value;


                    let description =
                        document.getElementById(
                            "description"
                        ).value;


                    let urgency =
                        document.getElementById(
                            "urgency"
                        ).value;


                    // Create complaint

                    let newComplaint = {

                        id:
                            Date.now(),

                        problem:
                            problem,

                        category:
                            category,

                        area:
                            area,

                        description:
                            description,

                        urgency:
                            urgency,

                        status:
                            "Reported",

                        supporters:
                            1

                    };


                    // Add to beginning

                    complaints.unshift(
                        newComplaint
                    );


                    // Save

                    saveComplaints();


                    // Show updated list

                    displayComplaints();


                    // Clear form

                    document
                        .getElementById(
                            "reportForm"
                        )
                        .reset();


                    alert(
                        "Your problem has been reported!"
                    );

                }
            );


        // -----------------------------------
        // Support a complaint
        // -----------------------------------

        function supportComplaint(id) {


            let complaint =
                complaints.find(
                    function(item) {

                        return item.id === id;

                    }
                );


            if (complaint) {

                complaint.supporters =
                    complaint.supporters + 1;


                saveComplaints();

                displayComplaints();

            }

        }


        // -----------------------------------
        // Show follow-up timeline
        // -----------------------------------

        function showTimeline(id) {


            let timeline =
                document.getElementById(
                    "timeline-" + id
                );


            if (
                timeline.style.display ===
                "block"
            ) {

                timeline.style.display =
                    "none";

            }

            else {

                timeline.style.display =
                    "block";

            }

        }


        // -----------------------------------
        // Update statistics
        // -----------------------------------

        function updateStatistics() {


            // Total complaints

            document.getElementById(
                "totalReports"
            ).innerText =
                complaints.length;


            // Complaints in progress

            let progress =
                complaints.filter(
                    function(item) {

                        return item.status ===
                            "In Progress";

                    }
                ).length;


            document.getElementById(
                "progressReports"
            ).innerText =
                progress;


            // Resolved complaints

            let resolved =
                complaints.filter(
                    function(item) {

                        return item.status ===
                            "Resolved";

                    }
                ).length;


            document.getElementById(
                "resolvedReports"
            ).innerText =
                resolved;


            // Total community support

            let support = 0;


            complaints.forEach(
                function(item) {

                    support =
                        support +
                        item.supporters;

                }
            );


            document.getElementById(
                "totalSupport"
            ).innerText =
                support;

        }


        // -----------------------------------
        // Search
        // -----------------------------------

        document
            .getElementById("search")
            .addEventListener(
                "input",
                function() {

                    displayComplaints();

                }
            );


        // -----------------------------------
        // Start the website
        // -----------------------------------

        displayComplaints();

        updateStatistics();


    </script>
